function cleanPoly(p) {
  let q = [];
  for (const v of p) { const l = q[q.length - 1]; if (!l || l[0] !== v[0] || l[1] !== v[1]) q.push(v); }
  if (q.length > 1 && q[0][0] === q[q.length - 1][0] && q[0][1] === q[q.length - 1][1]) q.pop();
  let changed = true;
  while (changed && q.length > 3) {
    changed = false;
    for (let i = 0; i < q.length; i++) {
      const a = q[(i + q.length - 1) % q.length], b = q[i], c = q[(i + 1) % q.length];
      if (Math.abs((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])) < 1e-12) { q.splice(i, 1); changed = true; break; }
    }
  }
  return q;
}
const area2 = p => { let s = 0; for (let i = 0; i < p.length; i++) { const a = p[i], b = p[(i + 1) % p.length]; s += a[0] * b[1] - b[0] * a[1]; } return s; };
const cross3 = (a, b, c) => (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
function inTri(p, a, b, c) {
  const d1 = cross3(a, b, p), d2 = cross3(b, c, p), d3 = cross3(c, a, p);
  return d1 > 1e-12 && d2 > 1e-12 && d3 > 1e-12;
}
function triangulate(p) {                 // ear clipping, p counter-clockwise, no holes
  if (p.length === 3) return [0, 1, 2];
  if (p.length === 4) {
    const r = cross3(p[0], p[1], p[2]) > 0 && cross3(p[0], p[2], p[3]) > 0 && !inTri(p[3], p[0], p[1], p[2]) && !inTri(p[1], p[0], p[2], p[3]);
    if (r) return [0, 1, 2, 0, 2, 3];
  }
  const idx = p.map((_, i) => i), out = [];
  let guard = 0;
  while (idx.length > 3 && guard++ < 5000) {
    let cut = false;
    for (let i = 0; i < idx.length; i++) {
      const ia = idx[(i + idx.length - 1) % idx.length], ib = idx[i], ic = idx[(i + 1) % idx.length];
      const a = p[ia], b = p[ib], c = p[ic];
      if (cross3(a, b, c) <= 1e-12) continue;
      let blocked = false;
      for (const j of idx) {
        if (j === ia || j === ib || j === ic) continue;
        const q = p[j];
        if ((q[0] === a[0] && q[1] === a[1]) || (q[0] === b[0] && q[1] === b[1]) || (q[0] === c[0] && q[1] === c[1])) continue;
        if (inTri(q, a, b, c)) { blocked = true; break; }
      }
      if (blocked) continue;
      out.push(ia, ib, ic); idx.splice(i, 1); cut = true; break;
    }
    if (!cut) break;
  }
  if (idx.length === 3) out.push(idx[0], idx[1], idx[2]);
  return out;
}
function addPrism(pos, lay, pts, L) {
  let p = cleanPoly(pts);
  if (p.length < 3) return;
  if (area2(p) < 0) p.reverse();
  const t = triangulate(p);
  const v = (q, h) => { pos.push(q[0], q[1], h); lay.push(L); };
  for (let i = 0; i < t.length; i += 3) {
    const a = p[t[i]], b = p[t[i + 1]], c = p[t[i + 2]];
    v(a, 1); v(b, 1); v(c, 1); v(a, 0); v(c, 0); v(b, 0);
  }
  for (let i = 0; i < p.length; i++) {
    const a = p[i], b = p[(i + 1) % p.length];
    v(a, 0); v(b, 0); v(b, 1); v(a, 0); v(b, 1); v(a, 1);
  }
}

function togglesUpTo(i, T) {               // number of toggles of net i at or before T
  let lo = togOff[i], hi = togOff[i + 1];

}

    P = persp(FOV, asp, near, far);
    P[8] -= viewShift();                // keep the die centred in the part of the stage the panel does not cover
  }
  gl.uniformMatrix4fv(U.uVP, false, new Float32Array(mul(P, V)));
  gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, actTex); gl.uniform1i(U.uAct, 0);
  gl.uniform1i(U.uSig, st.sig ? 1 : 0); gl.uniform1i(U.uHeat, st.heat ? 1 : 0); gl.uniform1f(U.uSelNet, st.selNet);
  gl.uniform3fv(U.uEye, e);
  gl.uniform1fv(U.uZ0, new Float32Array(LAYERS.map(l => l.z0 * st.thick + l.rank * st.sep)));
  gl.uniform1fv(U.uTh, new Float32Array(LAYERS.map(l => (l.z1 - l.z0) * st.thick)));
  gl.uniform1fv(U.uVis, new Float32Array(st.vis));
  gl.uniform3fv(U.uCol, colArr);
  gl.uniform1i(U.uPick, pick ? 1 : 0);
  gl.uniform1f(U.uSelId, st.selId);
  const secC = bnd[0] + st.secPos * W;
  gl.uniform3f(U.uSec, st.sec ? 1 : 0, secC - st.secW / 2, secC + st.secW / 2);
  gl.uniform1i(U.uTop, 0);
  gl.disableVertexAttribArray(5); gl.vertexAttrib1f(5, 0);
  for (const m of masters) {
    if (!m.nVert) continue;
    if (st.isolate >= 0 ? m.idx !== st.isolate : (!st.fill && m.filler)) continue;
    gl.uniform1f(U.uHiType, !pick && m.idx === st.hiType ? 1 : 0);
    gl.bindVertexArray(m.vao);
    gl.drawArraysInstanced(gl.TRIANGLES, 0, m.nVert, m.nInst);
  }
  gl.uniform1i(U.uTop, 1); gl.uniform1f(U.uHiType, 0);
  gl.uniform1f(U.uTopGeom, st.top && st.isolate < 0 ? 1 : 0);
  gl.bindVertexArray(topVao);
  gl.vertexAttrib1f(1, 0);
  gl.drawArraysInstanced(gl.TRIANGLES, 0, unitPos.length / 3, nTop);
  gl.bindVertexArray(null);
}
let last = performance.now(), interacting = false, playT = 0, secT = 3.5, scrubbing = false, actT = -1;
const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)");
function frame(t) {
  const dt = Math.min(0.05, (t - last) / 1000); last = t;
  sizeCanvas();
  if (st.play && !interacting) {
    // one loop of 18 s: stacked, layers lift apart, hold, settle back, while the camera circles
    playT += dt;
    const u = playT % 18, ease = x => (1 - Math.cos(Math.PI * Math.min(1, Math.max(0, x)))) / 2;
    const k = u < 2 ? 0 : u < 7 ? ease((u - 2) / 5) : u < 11 ? 1 : u < 16 ? 1 - ease((u - 11) / 5) : 0;
    st.sep = 9 * k; st.thick = 2.5;
    cam.th += dt * 0.3; cam.ph = 1.0 + 0.32 * k; cam.tgt[1] = stackMid();
    optSep.value = st.sep; optThick.value = st.thick; syncOut(); needs = true;
  } else if (st.spin && !interacting && !reduceMotion.matches) { cam.th += dt * 0.25; needs = true; }
  if (st.sig) {
    if (st.simPlay && !scrubbing) {
      st.simT += dt * 24 * Math.pow(2, st.speed);
      if (st.simT >= SEND) st.simT = 2 * SP;
    }
    if (st.simT !== actT) { updateAct(st.simT); actT = st.simT; syncSim(); needs = true; }
  }
  if (st.sec && st.secRun && !interacting) {
    secT += dt; st.secPos = 0.5 - 0.5 * Math.cos(secT * Math.PI * 2 / 14);
    optSecPos.value = st.secPos; syncSec(); needs = true;
  }
  if (needs) { draw(false); needs = false; }
  requestAnimationFrame(frame);
}

function pickAt(cx, cy) {
  ensurePick();
  draw(true);
  const rect = canvas.getBoundingClientRect();
  const x = Math.floor((cx - rect.left) / rect.width * canvas.width);
  const y = Math.floor((1 - (cy - rect.top) / rect.height) * canvas.height);
  const px = new Uint8Array(4);
  gl.bindFramebuffer(gl.FRAMEBUFFER, pickFbo.f);
  gl.readPixels(x, y, 1, 1, gl.RGBA, gl.UNSIGNED_BYTE, px);
  gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  dirty();
  return px[0] + px[1] * 256 + px[2] * 65536;
}

function pan(dx, dy) {
  let x, y;
  if (st.sig && st.topdown) { x = [1, 0, 0]; y = [0, 0, -1]; }
  else { const e = eye(), z = nrm(sub(e, cam.tgt)); x = nrm(crs([0, 1, 0], z)); y = crs(z, x); }
  const s = cam.r * 2 * Math.tan(FOV / 2) / canvas.clientHeight;
  for (let i = 0; i < 3; i++) cam.tgt[i] += (-x[i] * dx + y[i] * dy) * s;
}
canvas.addEventListener("pointermove", e => {
  if (!ptrs.has(e.pointerId)) return;
  const p = ptrs.get(e.pointerId), dx = e.clientX - p.x, dy = e.clientY - p.y;
  if (mode === "pinch" && ptrs.size === 2) {
    const before = [...ptrs.values()]; const mx0 = (before[0].x + before[1].x) / 2, my0 = (before[0].y + before[1].y) / 2;
    p.x = e.clientX; p.y = e.clientY;
    const [a, b] = [...ptrs.values()]; const d = Math.hypot(a.x - b.x, a.y - b.y);
    if (pinch0 > 0 && d > 0) cam.r = Math.min(SIZE * 6, Math.max(SIZE * 0.03, cam.r * pinch0 / d));
    pinch0 = d; pan((a.x + b.x) / 2 - mx0, (a.y + b.y) / 2 - my0); dirty(); return;
  }
  p.x = e.clientX; p.y = e.clientY;
  if (down && Math.hypot(e.clientX - down.x, e.clientY - down.y) > 5) down.moved = true;
  if (mode === "rot") { cam.th += dx * 0.006; cam.ph = Math.min(3.10, Math.max(0.03, cam.ph - dy * 0.006)); dirty(); }
  else if (mode === "pan") { pan(dx, dy); dirty(); }
});
function up(e) {

    down = null; mode = null; interacting = false; canvas.classList.remove("dragging");
  } else if (ptrs.size === 1) { mode = st.sig && st.topdown ? "pan" : "rot"; }
}
